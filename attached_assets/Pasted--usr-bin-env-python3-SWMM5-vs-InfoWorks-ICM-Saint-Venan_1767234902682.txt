# -----------------------------
# Rendering helpers
# -----------------------------
def wrap(text: str, width: int = 92, subsequent_indent: str = "  ") -> str:
    return textwrap.fill(text, width=width, subsequent_indent=subsequent_indent)


def bullet_list(items: List[str], width: int = 92) -> str:
    out: List[str] = []
    for i in items:
        wrapped = textwrap.fill(i, width=width, initial_indent="• ", subsequent_indent="  ")
        out.append(wrapped)
    return "\n".join(out)


def hr(char: str = "-", n: int = 92) -> str:
    return char * n


def ascii_table(headers: List[str], rows: List[List[str]], max_width: int = 120) -> str:
    """
    Minimal ASCII table printer (no external deps). Wraps cell contents.
    """
    col_count = len(headers)
    assert all(len(r) == col_count for r in rows), "Row/column mismatch."

    widths = [len(h) for h in headers]
    for r in rows:
        for c, cell in enumerate(r):
            widths[c] = max(widths[c], min(max(len(line) for line in cell.splitlines()), max_width))

    total = sum(widths) + 3 * (col_count - 1)
    if total > max_width:
        min_w = [min(w, 18) for w in widths]
        widths = min_w
        remaining = max_width - (sum(widths) + 3 * (col_count - 1))
        if remaining > 0:
            gi = 0
            while remaining > 0 and gi < 2000:
                idx = gi % col_count
                widths[idx] += 1
                remaining -= 1
                gi += 1

    def wrap_cell(s: str, w: int) -> List[str]:
        parts: List[str] = []
        for para in s.splitlines():
            para = para.rstrip()
            if not para:
                parts.append("")
                continue
            wrapped = textwrap.wrap(para, width=w) or [""]
            parts.extend(wrapped)
        return parts or [""]

    lines: List[str] = []
    sep = "+".join(["-" * (w + 2) for w in widths])
    sep = f"+{sep}+"

    def row_line(cells: List[str]) -> List[str]:
        wrapped_cols = [wrap_cell(cells[i], widths[i]) for i in range(col_count)]
        height = max(len(wc) for wc in wrapped_cols)
        out_lines: List[str] = []
        for r_i in range(height):
            pieces = []
            for c in range(col_count):
                txt = wrapped_cols[c][r_i] if r_i < len(wrapped_cols[c]) else ""
                pieces.append(f" {txt.ljust(widths[c])} ")
            out_lines.append("|" + "|".join(pieces) + "|")
        return out_lines

    lines.append(sep)
    lines.extend(row_line(headers))
    lines.append(sep)
    for r in rows:
        lines.extend(row_line(r))
        lines.append(sep)
    return "\n".join(lines)


def build_side_by_side_rows() -> List[List[str]]:
    rows: List[List[str]] = []
    for key, label in TOPIC_ORDER:
        sw = "\n".join([f"- {x}" for x in KB["swmm5"]["topics"][key]])
        icm = "\n".join([f"- {x}" for x in KB["icm"]["topics"][key]])
        rows.append([label, sw, icm])
    return rows


def export_markdown(path: str) -> None:
    now = datetime.utcnow().strftime("%Y-%m-%d %H:%M UTC")
    lines: List[str] = []
    lines.append("# SWMM5 vs InfoWorks ICM — Saint‑Venant Solver Comparison")
    lines.append("")
    lines.append(f"_Generated: {now}_")
    lines.append("")
    lines.append("## High-level summary")
    lines.append("")
    lines.append(f"**SWMM 5:** {KB['swmm5']['tagline']}")
    lines.append("")
    lines.append(f"**InfoWorks ICM:** {KB['icm']['tagline']}")
    lines.append("")
    lines.append("## Detailed comparison by topic")
    lines.append("")
    for key, label in TOPIC_ORDER:
        lines.append(f"### {label}")
        lines.append("")
        lines.append("**SWMM 5**")
        for b in KB["swmm5"]["topics"][key]:
            lines.append(f"- {b}")
        lines.append("")
        lines.append("**InfoWorks ICM**")
        for b in KB["icm"]["topics"][key]:
            lines.append(f"- {b}")
        lines.append("")
    lines.append("## Sources")
    lines.append("")
    for model_key in ("swmm5", "icm"):
        lines.append(f"### {KB[model_key]['product']}")
        for s in KB[model_key]["sources"]:
            lines.append(f"- {s['label']} — {s['url']}")
            if s.get("notes"):
                lines.append(f"  - Notes: {s['notes']}")
        lines.append("")
    with open(path, "w", encoding="utf-8") as f:
        f.write("\n".join(lines))


def export_json(path: str) -> None:
    with open(path, "w", encoding="utf-8") as f:
        json.dump(KB, f, indent=2)

# CLI / Interactive UI

def print_header() -> None:
    print(hr("="))
    print("SWMM5 vs InfoWorks ICM — Saint‑Venant (Dynamic Wave) Solver Comparison")
    print(hr("="))
    print(wrap("This program provides a structured compare/contrast of how EPA SWMM 5 and InfoWorks ICM "
               "solve the 1D Saint‑Venant equations for unsteady flow in drainage networks."))
    print()
    print("High-level taglines:")
    print(f"• SWMM 5: {KB['swmm5']['tagline']}")
    print(f"• InfoWorks ICM: {KB['icm']['tagline']}")
    print()


def show_summary_table() -> None:
    headers = ["Topic", "SWMM 5 (Dynamic Wave)", "InfoWorks ICM (1D)"]
    rows = build_side_by_side_rows()
    print(ascii_table(headers, rows, max_width=120))


def show_topic(topic_key: str) -> None:
    label_map = {k: v for k, v in TOPIC_ORDER}
    label = label_map.get(topic_key, topic_key)

    print(hr())
    print(label)
    print(hr())
    print()
    print("SWMM 5")
    print(hr("-", 6))
    print(bullet_list(KB["swmm5"]["topics"][topic_key]))
    print()
    print("InfoWorks ICM")
    print(hr("-", 12))
    print(bullet_list(KB["icm"]["topics"][topic_key]))
    print()


def list_topics() -> None:
    print("Topics:")
    for i, (key, label) in enumerate(TOPIC_ORDER, start=1):
        print(f"  {i}. {label} ({key})")


def show_sources() -> None:
    print(hr())
    print("Sources (for traceability)")
    print(hr())
    for model_key in ("swmm5", "icm"):
        print()
        print(KB[model_key]["product"])
        print(hr("-", len(KB[model_key]["product"])))
        for s in KB[model_key]["sources"]:
            print(f"• {s['label']}")
            print(f"  {s['url']}")
            if s.get("notes"):
                print(wrap(f"Notes: {s['notes']}", subsequent_indent="        "))
    print()


def interactive() -> None:
    print_header()
    while True:
        print(hr())
        print("Menu")
        print(hr())
        print("1) View side-by-side summary table")
        print("2) View a topic (deep dive)")
        print("3) Export to Markdown (comparison.md)")
        print("4) Export to JSON (comparison.json)")
        print("5) Show sources")
        print("0) Exit")
        choice = input("\nSelect an option: ").strip()

        if choice == "1":
            print()
            show_summary_table()
            print()

        elif choice == "2":
            print()
            list_topics()
            sel = input("\nEnter topic number (or key): ").strip()
            if sel.isdigit():
                idx = int(sel)
                if 1 <= idx <= len(TOPIC_ORDER):
                    key = TOPIC_ORDER[idx - 1][0]
                    show_topic(key)
                else:
                    print("Invalid topic number.")
            else:
                key = sel
                if key in {k for k, _ in TOPIC_ORDER}:
                    show_topic(key)
                else:
                    print("Unknown topic key.")
            print()

        elif choice == "3":
            path = "comparison.md"
            export_markdown(path)
            print(f"\nWrote {path} to: {os.path.abspath(path)}\n")

        elif choice == "4":
            path = "comparison.json"
            export_json(path)
            print(f"\nWrote {path} to: {os.path.abspath(path)}\n")

        elif choice == "5":
            print()
            show_sources()

        elif choice == "0":
            print("\nDone.")
            break

        else:
            print("Unknown option. Try again.")


def main() -> None:
    parser = argparse.ArgumentParser(
        description="Compare SWMM5 Dynamic Wave vs InfoWorks ICM 1D Saint-Venant solver approaches."
    )
    parser.add_argument("--table", action="store_true", help="Print the side-by-side summary table and exit.")
    parser.add_argument("--topic", type=str, help="Print a single topic by key (e.g., time_integration) and exit.")
    parser.add_argument("--list-topics", action="store_true", help="List available topics and exit.")
    parser.add_argument("--sources", action="store_true", help="Print sources and exit.")
    parser.add_argument("--export-md", metavar="PATH", help="Export comparison to a Markdown file.")
    parser.add_argument("--export-json", metavar="PATH", help="Export the underlying knowledge base to JSON.")
    args = parser.parse_args()

    if args.list_topics:
        list_topics()
        return

    if args.sources:
        show_sources()
        return

    if args.export_md:
        export_markdown(args.export_md)
        print(f"Wrote {args.export_md}")
        return

    if args.export_json:
        export_json(args.export_json)
        print(f"Wrote {args.export_json}")
        return

    if args.table:
        print_header()
        show_summary_table()
        return

    if args.topic:
        if args.topic not in KB["swmm5"]["topics"]:
            print(f"Unknown topic: {args.topic}")
            list_topics()
            raise SystemExit(2)
        print_header()
        show_topic(args.topic)
        return

    # Default: interactive
    interactive()


if __name__ == "__main__":
    main()
